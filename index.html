function calculateLineOut() { const targetDepth = parseFloat(document.getElementById('desiredDepth').value); const lure = document.getElementById('lureType').value; const lineLb = parseFloat(document.getElementById('lineType').value); const rodHeight = parseFloat(document.getElementById('rodHeight').value); // 1. Calculate effective depth required at lure level to compensate for rod height // Offset: +0.667 ft of lure depth required for every 1.0 ft of rod height above 0.5 ft const heightOffset = (rodHeight - 0.5) * 0.6667; const effectiveDepthNeeded = targetDepth + heightOffset; // 2. Line Diameter Factor (Base 1.0 = 10lb Mono) // 8lb mono = 0.93 drag factor; 20lb mono = 1.18 drag factor let lineDragFactor = 1.0; if (lineLb === 8) lineDragFactor = 0.93; if (lineLb === 20) lineDragFactor = 1.18; let finalLineOut = 0; let maxDepthPossible = 0; if (lure === 'PICO') { maxDepthPossible = (23.0 / lineDragFactor) - heightOffset; if (effectiveDepthNeeded <= 23.5) { // Quadratic model fitted precisely to Leland's PICO 10lb Mono benchmarks (40'->13', 60'->16.5', 80'->20', 100'->23') const D = effectiveDepthNeeded; const baseLineOut = (0.2083 * D * D) - (1.50 * D) + 25.0; finalLineOut = baseLineOut * lineDragFactor; } } else if (lure === 'Lucas') { maxDepthPossible = (24.0 / lineDragFactor) - heightOffset; if (effectiveDepthNeeded <= 24.5) { // Quadratic model fitted precisely to Leland's Lucas 10lb Mono benchmarks (40'->15', 60'->18', 80'->21', 100'->24') const D = effectiveDepthNeeded; const baseLineOut = (0.1852 * D * D) - (1.33 * D) + 18.0; finalLineOut = baseLineOut * lineDragFactor; } } // UI Updates const warningElement = document.getElementById('depthWarning'); if (finalLineOut > 0 && targetDepth <= maxDepthPossible && finalLineOut <= 150) { document.getElementById('lineOutResult').innerText = Math.round(finalLineOut) + " ft"; warningElement.style.display = 'none'; } else { document.getElementById('lineOutResult').innerText = "N/A"; warningElement.innerText = `* Selected depth exceeds max reach (~${Math.floor(maxDepthPossible)} ft) for this setup.`; warningElement.style.display = 'block'; } } calculateLineOut();
